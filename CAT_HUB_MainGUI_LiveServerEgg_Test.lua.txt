    local function getMyEggs()
        local t = {}
        if not Networking or not Assets then return t end
        pcall(function()
            local res = Networking:FindFirstChild("RF/EggWorld/AskFieldEggSnapshot"):InvokeServer()
            if res and res.Records then
                for _, rc in pairs(res.Records) do
                    local e = Assets.Directory[tostring(rc.AssetCategory)]
                    if e then
                        local s = tonumber(rc.AssetScale) or 1
                        local earn = (tonumber(e.EarningRate) or 0) * (s ^ 1.5)
                        table.insert(t, {name = e.DisplayName or rc.AssetCategory, rarity = e.Rarity and tostring(e.Rarity.DisplayName) or "Unk", scale = s, valueNum = earn, value = earn >= 1e9 and string.format("$%.2fB/s", earn/1e9) or (earn >= 1e6 and string.format("$%.2fM/s", earn/1e6) or string.format("$%.0f/s", earn)), image = "rbxassetid://" .. tostring(e.Icon)})
                    end
                end
            end
        end)
        table.sort(t, function(a,b) return a.scale > b.scale end)
        return t
    end

    local function refreshServ()
        for _, c in ipairs(sScr:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
        local eggs = getMyEggs()
        table.sort(eggs, function(a, b) return (tonumber(a.valueNum) or 0) > (tonumber(b.valueNum) or 0) end)
        if #eggs == 0 then
            local empty = Instance.new("TextLabel", sScr); empty.Size = UDim2.new(1, -10, 0, 42); empty.BackgroundTransparency = 1; empty.Text = "No server eggs found / snapshot unavailable"; empty.TextColor3 = TXT_SUB; empty.Font = Enum.Font.Gotham; empty.TextSize = 12
        else
            for _, egg in ipairs(eggs) do mkRow(egg, sScr, false) end
        end
        sScr.CanvasSize = UDim2.new(0, 0, 0, sLay.AbsoluteContentSize.Y + 10)
    end

    local function requestLiveEggs()
        if type(requestFunc) ~= "function" then
            warn("[CAT HUB] No compatible HTTP request function.")
            return {}
        end
        if CAT_HUB_API_URL:find("YOUR-RENDER-SERVICE", 1, true) then
            warn("[CAT HUB] Set CAT_HUB_API_URL to your Render service URL.")
            return {}
        end

        local ok, response = pcall(function()
            return requestFunc({
                Url = CAT_HUB_API_URL .. "/eggs",
                Method = "GET",
                Headers = {
                    ["Accept"] = "application/json"
                    -- Add x-api-key only for private/local testing.
                    -- Never distribute a shared secret inside a public Lua script.
                }
            })
        end)

        if not ok or not response or not response.Body then
            warn("[CAT HUB] API request failed:", tostring(response))
            return {}
        end

        local status = tonumber(response.StatusCode or response.Status)
        if status and status >= 400 then
            warn("[CAT HUB] API returned HTTP", status,
                "- if /eggs requires x-api-key, configure a safe read-only public endpoint on the backend.")
            return {}
        end

        local decodedOk, data = pcall(function()
            return HttpService:JSONDecode(response.Body)
        end)
        if not decodedOk or type(data) ~= "table" then
            warn("[CAT HUB] API returned invalid JSON.")
            return {}
        end

        -- Support either a direct array or common API wrappers.
        local eggs = data
        if type(data.eggs) == "table" then
            eggs = data.eggs
        elseif type(data.data) == "table" then
            eggs = data.data
        elseif type(data.liveEggs) == "table" then
            eggs = data.liveEggs
        end

        if type(eggs) ~= "table" then
            warn("[CAT HUB] API JSON did not contain an egg list.")
            return {}
        end
        return eggs
    end

    local previousLive = {}
    local watchedEggName = ""
    local function notify(title, message)
        pcall(function()
            StarterGui:SetCore("SendNotification", {Title = title, Text = message, Duration = 5})
        end)
    end

    local function fetchLive()
        for _, child in ipairs(lScr:GetChildren()) do
            if child:IsA("Frame") then child:Destroy() end
        end

        local eggs = requestLiveEggs()
        local query = string.lower(sBox.Text or "")
        local shown = 0
        local currentLive = {}
        local watchQuery = string.lower(tostring(watchedEggName or ""))

        table.sort(eggs, function(a, b)
            return (tonumber(a.valueNum) or 0) > (tonumber(b.valueNum) or 0)
        end)

        for _, egg in ipairs(eggs) do
            local eggName = tostring(egg.name or "Unknown")
            local eggId = tostring(egg.serverId or "") .. "|" .. eggName
            currentLive[eggId] = egg
            if watchQuery ~= "" and string.find(string.lower(eggName), watchQuery, 1, true) and not previousLive[eggId] then
                notify("CAT HUB • Egg Found", eggName .. " is currently detected!")
            end
            local nameMatch = query == "" or string.find(string.lower(tostring(egg.name or "")), query, 1, true)
            local rarityMatch = fBtn.Text == "Rarity Filter" or string.find(
                string.lower(tostring(egg.rarity or "")),
                string.lower(fBtn.Text),
                1,
                true
            )
            if nameMatch and rarityMatch then
                mkRow(egg, lScr, true)
                shown = shown + 1
            end
        end

        for eggId, oldEgg in pairs(previousLive) do
            if not currentLive[eggId] then
                local oldName = tostring(oldEgg.name or "Egg")
                if watchQuery ~= "" and string.find(string.lower(oldName), watchQuery, 1, true) then
                    notify("CAT HUB • Egg Missing", oldName .. " is no longer detected. It may have moved, despawned, or the feed changed.")
                end
            end
        end
        previousLive = currentLive
        lTxt.Text = "Egg Live • " .. tostring(shown)
        lScr.CanvasSize = UDim2.new(0, 0, 0, lLay.AbsoluteContentSize.Y + 8)
        if shown == 0 then
            warn("[CAT HUB] No eggs returned or matched. Check API URL, response format, and endpoint access.")
        end
    end

    local rarityFilters = {"Any", "Common", "Uncommon", "Rare", "Epic", "Legendary", "Mythic", "Cosmic", "Secret", "Eternal", "Divine"}
    local rarityFilterIndex = 1
    fBtn.MouseButton1Click:Connect(function()
        rarityFilterIndex = rarityFilterIndex % #rarityFilters + 1
        fBtn.Text = rarityFilterIndex == 1 and "Rarity Filter" or rarityFilters[rarityFilterIndex]
        fetchLive()
    end)
    sBox:GetPropertyChangedSignal("Text"):Connect(function()
        watchedEggName = sBox.Text
        if fLive.Visible then fetchLive() end
    end)

    local snipe = false
    tgBtn.MouseButton1Click:Connect(function()
        if snipe then snipe = false; tgBtn.Text = "OFF"; tgBtn.TextColor3 = TXT_SUB; tgSt.Color = TXT_SUB; tgBtn.BackgroundColor3 = BG; return end
        snipe = true; tgBtn.Text = "ON"; tgBtn.TextColor3 = GREEN; tgSt.Color = GREEN; tgBtn.BackgroundColor3 = Color3.fromRGB(0, 50, 0)
        local kgV = parseKg(kBox.Text); local rarT = rBtn.Text
        task.spawn(function()
            while snipe do
                task.wait(3)
                local eggs = requestLiveEggs()
                for _, e in ipairs(eggs) do
                    local mR = (rarT == "Any" or string.find(string.lower(tostring(e.rarity or "")), string.lower(rarT), 1, true))
                    local scaleValue = tonumber(e.scale) or 0
                    local mK = (kgV == 0 or scaleValue >= kgV)
                    if mR and mK and e.serverId then
                        snipe = false
                        tgBtn.Text = "FOUND"
                        tgSt.Color = CYAN
                        pcall(function()
                            setclipboard("roblox://placeId=" .. tostring(e.placeId or game.PlaceId) .. "&gameInstanceId=" .. tostring(e.serverId))
                        end)
                        task.wait(2)
                        tgBtn.Text = "OFF"
                        tgBtn.TextColor3 = TXT_SUB
                        tgSt.Color = TXT_SUB
                        tgBtn.BackgroundColor3 = BG
                        break
                    end
                end
            end
        end)
    end)

    local function setTab(n, b, s)
        if aTab and aStr then aTab.BackgroundColor3 = CONTENT; aTab.TextColor3 = TXT_SUB; aStr.Transparency = 1 end
        if b and s then b.BackgroundColor3 = BOX_OUT; b.TextColor3 = CYAN; s.Transparency = 0; aTab = b; aStr = s end
        fHome.Visible = (n == "Home"); fAuto.Visible = (n == "Auto"); fLive.Visible = (n == "Live"); fServ.Visible = (n == "Serv")
        if n == "Serv" then pcall(refreshServ) end; if n == "Live" then pcall(fetchLive) end
    end

    tbHome.MouseButton1Click:Connect(function() setTab("Home", tbHome, stHome) end)
    tbAuto.MouseButton1Click:Connect(function() setTab("Auto", tbAuto, stAuto) end)
    tbLive.MouseButton1Click:Connect(function() setTab("Live", tbLive, stLive) end)
    tbServ.MouseButton1Click:Connect(function() setTab("Serv", tbServ, stServ) end)
    setTab("Home", tbHome, stHome)

    task.spawn(function()
        while task.wait(10) do
            if fLive.Visible then
                pcall(fetchLive)
            end
        end
    end)
