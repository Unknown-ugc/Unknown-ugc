local function eggMatches(egg)
    local query = string.lower(sBox.Text or "")
    local name = string.lower(tostring(egg.name or ""))
    local rarity = string.lower(tostring(egg.rarity or ""))
    local qMatch = query == "" or string.find(name, query, 1, true) ~= nil
    local wanted = rarityOptions[selectedRarity]
    return qMatch and (wanted == "Any" or rarity == string.lower(wanted))
end
local function redrawLive()
    for _, child in ipairs(lScr:GetChildren()) do
        if child:IsA("Frame") or child.Name == "EmptyState" then child:Destroy() end
    end
    local shown = 0
    for _, egg in ipairs(liveEggCache) do
        if eggMatches(egg) then mkRow(egg, lScr, true); shown += 1 end
    end
    if shown == 0 then
        local empty = Instance.new("TextLabel", lScr)
        empty.Name = "EmptyState"; empty.Size = UDim2.new(1, -12, 0, 52)
        empty.BackgroundColor3 = BOX_IN; empty.TextColor3 = TXT_SUB
        empty.Font = Enum.Font.Gotham; empty.TextSize = 12; empty.TextWrapped = true
        empty.Text = (#liveEggCache == 0) and "No live eggs yet — turn on ALT detector and wait for a detection." or "No eggs match this search/filter."

    end
    lTxt.Text = "LIVE EGG  •  " .. tostring(shown) .. " SHOWN / " .. tostring(#liveEggCache) .. " TOTAL"
    lScr.CanvasSize = UDim2.new(0, 0, 0, math.max(60, shown * 61))
end

    local function getMyEggs()
        local t = {}
        if not Networking or not Assets then return t end
        pcall(function()
            local res = Networking:FindFirstChild("RF/EggWorld/AskFieldEggSnapshot"):InvokeServer()
            if res and res.Records then
                for _, rc in pairs(res.Records) do
                    local e = Assets.Directory[tostring(rc.AssetCategory)]
                    if e then
                        local s = tonumber(rc.AssetScale) or 1
                        table.insert(t, {name = e.DisplayName or rc.AssetCategory, rarity = e.Rarity and tostring(e.Rarity.DisplayName) or "Unk", scale = s, value = string.format("$%.1fM", ((tonumber(e.EarningRate) or 0) * (s ^ 1.5))/1e6), image = "rbxassetid://" .. tostring(e.Icon)})
                    end
                end
            end
        end)
        table.sort(t, function(a,b) return (tonumber(a.valueNum) or tonumber(a.scale) or 0) > (tonumber(b.valueNum) or tonumber(b.scale) or 0) end)
        return t
    end

    local function refreshServ()
        for _, c in ipairs(sScr:GetChildren()) do if c:IsA("Frame") or c:IsA("TextLabel") then c:Destroy() end end
        local eggs = getMyEggs()
        for _, egg in ipairs(eggs) do mkRow(egg, sScr, false) end
        if #eggs == 0 then
            local empty = Instance.new("TextLabel", sScr)
            empty.Size = UDim2.new(1, -12, 0, 52); empty.BackgroundColor3 = BOX_IN
            empty.Text = "No server eggs returned. The game's snapshot remote or Assets mapping may differ."
            empty.TextColor3 = TXT_SUB; empty.TextWrapped = true; empty.Font = Enum.Font.Gotham; empty.TextSize = 12
        end
        sScr.CanvasSize = UDim2.new(0, 0, 0, math.max(60, #eggs * 61))
    end

    local function fetchLive()
        if not requestFunc then
            liveEggCache = {}
            redrawLive()
            return
        end
        local ok, r = pcall(function()
            return requestFunc({ Url = API_EGGS_URL, Method = "GET", Headers = { ["Accept"] = "application/json" } })
        end)
        if not ok or not r or (r.StatusCode and r.StatusCode >= 400) or not r.Body then
            liveEggCache = {}
            redrawLive()
            return
        end
        local decodedOk, dec = pcall(function() return HttpService:JSONDecode(r.Body) end)
        if not decodedOk or type(dec) ~= "table" then liveEggCache = {}; redrawLive(); return end
        local eggs = dec
        if type(dec.eggs) == "table" then eggs = dec.eggs
        elseif type(dec.data) == "table" then eggs = dec.data end
        if type(eggs) ~= "table" then eggs = {} end
        liveEggCache = {}
        for _, egg in ipairs(eggs) do
            if type(egg) == "table" then
                local normalized = {}
                for k, v in pairs(egg) do normalized[k] = v end
                normalized.image = normalized.image or normalized.icon or ""
                normalized.scale = tonumber(normalized.scale) or 0
                normalized.valueNum = tonumber(normalized.valueNum) or 0
                table.insert(liveEggCache, normalized)
            end
        end
        table.sort(liveEggCache, function(a,b) return (tonumber(a.valueNum) or 0) > (tonumber(b.valueNum) or 0) end)
        glw.BackgroundColor3 = GREEN
        liveGlow.Color = GREEN
        redrawLive()
    end

    local snipe = false
    tgBtn.MouseButton1Click:Connect(function()
        if snipe then snipe = false; tgBtn.Text = "OFF"; tgBtn.TextColor3 = TXT_SUB; tgSt.Color = TXT_SUB; tgBtn.BackgroundColor3 = BG; return end
        snipe = true; tgBtn.Text = "ON"; tgBtn.TextColor3 = GREEN; tgSt.Color = GREEN; tgBtn.BackgroundColor3 = Color3.fromRGB(0, 50, 0)
        local kgV = parseKg(kBox.Text); local rarT = rBtn.Text
        task.spawn(function()
            while snipe do
                task.wait(3)
                local ok, r = pcall(function() return requestFunc({ Url = API_EGGS_URL, Method = "GET" }) end)
                if ok and r and r.Body then
                    local dec = HttpService:JSONDecode(r.Body)
                    if type(dec) == "table" then
                        local eggs = dec
                        if type(dec.eggs) == "table" then eggs = dec.eggs elseif type(dec.data) == "table" then eggs = dec.data end
                        for _, e in ipairs(eggs) do
                            local mR = (rarT == "Any" or string.find(string.lower(e.rarity or ""), string.lower(rarT)))
                            local mK = false
                            if kgV == 0 then mK = true else local mag = 10 ^ math.floor(math.log10(kgV)); mK = ((e.scale or 0) >= kgV and (e.scale or 0) <= (math.floor(kgV / mag) + 1) * mag - 1) end
                            if mR and mK then
                                snipe = false; tgBtn.Text = "FOUND"; tgSt.Color = CYAN
                                pcall(function() setclipboard("roblox://placeId=" .. game.PlaceId .. "&gameInstanceId=" .. e.serverId) end)
                                task.wait(2); tgBtn.Text = "OFF"; tgBtn.TextColor3 = TXT_SUB; tgSt.Color = TXT_SUB; tgBtn.BackgroundColor3 = BG; break
                            end
                        end
                    end
                end
            end
        end)
    end)

    local function setTab(n, b, s)
        if aTab and aStr then aTab.BackgroundColor3 = CONTENT; aTab.TextColor3 = TXT_SUB; aStr.Transparency = 1 end
        if b and s then b.BackgroundColor3 = BOX_OUT; b.TextColor3 = CYAN; s.Transparency = 0; aTab = b; aStr = s end
        fHome.Visible = (n == "Home"); fAuto.Visible = (n == "Auto"); fLive.Visible = (n == "Live"); fServ.Visible = (n == "Serv")
        if n == "Serv" then pcall(refreshServ) end; if n == "Live" then pcall(fetchLive) end
    end

    tbAuto.MouseButton1Click:Connect(function() setTab("Auto", tbAuto, stAuto) end)
    tbLive.MouseButton1Click:Connect(function() setTab("Live", tbLive, stLive) end)
    tbServ.MouseButton1Click:Connect(function() setTab("Serv", tbServ, stServ) end)
    setTab("Home", tbHome, stHome)

    task.spawn(function() while task.wait(5) do if fLive.Visible then pcall(fetchLive) end end end)
end
