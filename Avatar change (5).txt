local function makeDraggable(object)
    local dragging = false
    local dragStart, startPosition
    local moved = false

    object.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPosition = object.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if math.abs(delta.X) > 4 or math.abs(delta.Y) > 4 then
                moved = true
            end
            object.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)

    return function() return moved end
end

local function getCharacter()
    return LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()

local function getHRP(character)
    return character and character:FindFirstChild("HumanoidRootPart")
end

local function getOverlayFolder()
    local camera = workspace.CurrentCamera
    if not camera then return nil end
    local folder = camera:FindFirstChild("_ChrisAvatarOverlay")
    if not folder then
        folder = Instance.new("Folder")
        folder.Name = "_ChrisAvatarOverlay"
        folder.Parent = camera
    end
    return folder
end

local function stopMirror()
    if Overlay.renderConn then
        Overlay.renderConn:Disconnect()
        Overlay.renderConn = nil
    end
end

local function clearMotorPairs()
    table.clear(Overlay.motorPairs)
end

local function restoreRealAvatar()
    for instance, oldTransparency in pairs(Overlay.hiddenParts) do
        if instance and instance.Parent then
            if instance:IsA("BasePart") then
                instance.LocalTransparencyModifier = oldTransparency
            elseif instance:IsA("Decal") or instance:IsA("Texture") then
                instance.Transparency = oldTransparency
            end
        end
    end
    table.clear(Overlay.hiddenParts)
end

local function hideRealAvatar(character)
    if not character then return end
    table.clear(Overlay.hiddenParts)
    for _, instance in ipairs(character:GetDescendants()) do
        if instance:IsA("BasePart") then
            Overlay.hiddenParts[instance] = instance.LocalTransparencyModifier
            instance.LocalTransparencyModifier = 1
        elseif instance:IsA("Decal") or instance:IsA("Texture") then
            Overlay.hiddenParts[instance] = instance.Transparency
            instance.Transparency = 1
        end
    end
end

local function prepareModel(model)
    for _, instance in ipairs(model:GetDescendants()) do
        if instance:IsA("Script") or instance:IsA("LocalScript")
            or instance:IsA("ModuleScript") then
            pcall(function() instance:Destroy() end)
        elseif instance:IsA("BasePart") then
            instance.CanCollide = false
            instance.CanTouch = false
            instance.CanQuery = false
            instance.Massless = true
        end
    end

    local humanoid = model:FindFirstChildOfClass("Humanoid")
    if humanoid then
        humanoid.PlatformStand = true
        humanoid.AutoRotate = false
        humanoid.Sit = true
        pcall(function()
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Physics, false)
        end)
    end
end

local function motorKey(motor)
    local part0 = motor.Part0 and motor.Part0.Name or ""
    local part1 = motor.Part1 and motor.Part1.Name or ""
    return motor.Name .. "|" .. part0 .. ">" .. part1
end

local function buildMotorPairs(sourceCharacter, targetModel)
    clearMotorPairs()
    local sourceByKey, targetByKey = {}, {}
    local sourceByName, targetByName = {}, {}

    for _, instance in ipairs(sourceCharacter:GetDescendants()) do
        if instance:IsA("Motor6D") then
            sourceByKey[motorKey(instance)] = instance
            if not sourceByName[instance.Name] then
                sourceByName[instance.Name] = instance
            end
        end
    end

    for _, instance in ipairs(targetModel:GetDescendants()) do
        if instance:IsA("Motor6D") then
            targetByKey[motorKey(instance)] = instance
            if not targetByName[instance.Name] then
                targetByName[instance.Name] = instance
            end
        end
    end

    local matched = {}
    for key, sourceMotor in pairs(sourceByKey) do
        local targetMotor = targetByKey[key]
        if targetMotor then
            table.insert(Overlay.motorPairs, { src = sourceMotor, tgt = targetMotor })
            matched[targetMotor] = true
        end
    end
    for name, sourceMotor in pairs(sourceByName) do
        local targetMotor = targetByName[name]
        if targetMotor and not matched[targetMotor] then
            table.insert(Overlay.motorPairs, { src = sourceMotor, tgt = targetMotor })
            matched[targetMotor] = true
        end
    end
end

local function startMirror()
    stopMirror()
    Overlay.renderConn = RunService.RenderStepped:Connect(function()
        local visual = Overlay.model
        local visualHRP = Overlay.hrp
        if not visual or not visualHRP
            or not visual.Parent or not visualHRP.Parent then return end

        local realCharacter = getCharacter()
        local realHRP = getHRP(realCharacter)
        if not realHRP then return end

        if (visualHRP.Position - realHRP.Position).Magnitude > 0.05 then
            visual:PivotTo(realHRP.CFrame)
        end

        for _, pair in ipairs(Overlay.motorPairs) do
            local s, t = pair.src, pair.tgt
            if s and t and s.Parent and t.Parent then
                t.Transform = s.Transform
            end
        end
    end)
end

local function removeAvatar()
    stopMirror()
    clearMotorPairs()
    if Overlay.model then
        pcall(function() Overlay.model:Destroy() end)
    end
    Overlay.model = nil
    Overlay.hrp = nil
    Overlay.weld = nil
    restoreRealAvatar()
    Overlay.lastUserId = nil
    Overlay.lastUsername = nil
end

local function changeAvatarTo(userId, name)
    if Busy then return false, "Please wait..." end
    if not userId then return false, "No user loaded." end

    Busy = true
    removeAvatar()

    local realCharacter = getCharacter()
    local realHRP = getHRP(realCharacter)
    if not realHRP then
        Busy = false
        return false, "Character not ready."
    end

    local ok, targetModel = pcall(function()
        return Players:CreateHumanoidModelFromUserIdAsync(userId)
    end)
    if not ok or not targetModel then
        Busy = false
        return false, "Failed to load avatar."
    end

    prepareModel(targetModel)
    targetModel.Name = "ChrisAvatarOverlay"

    local folder = getOverlayFolder()
    if not folder then
        targetModel:Destroy()
        Busy = false
        return false, "Camera not ready."
    end

    targetModel.Parent = folder

    local targetHRP = targetModel:FindFirstChild("HumanoidRootPart")
    if not targetHRP then
        targetModel:Destroy()
        Busy = false
        return false, "Avatar has no HumanoidRootPart."
    end

    targetModel:PivotTo(realHRP.CFrame)

    local weld = Instance.new("WeldConstraint")
    weld.Name = "ChrisAvatarWeld"
    weld.Part0 = targetHRP
    weld.Part1 = realHRP
    weld.Parent = targetHRP

    targetHRP.CanCollide = false
    targetHRP.CanTouch = false
    targetHRP.CanQuery = false
    targetHRP.Massless = true

    Overlay.model = targetModel
    Overlay.hrp = targetHRP
    Overlay.weld = weld
    Overlay.lastUserId = userId
    Overlay.lastUsername = name

    buildMotorPairs(realCharacter, targetModel)
    startMirror()
    hideRealAvatar(realCharacter)

    Busy = false
    return true, "Avatar changed: @" .. name
end

local function resolveUser(text)
    text = tostring(text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if text == "" then return nil, "Enter a username or UserId." end

    local id, name
    local asNumber = tonumber(text)

    if asNumber then
        id = math.floor(asNumber)
        local ok, n = pcall(function()
            return Players:GetNameFromUserIdAsync(id)
        end)
        if ok and n then name = n end
    else
        local ok, i = pcall(function()
            return Players:GetUserIdFromNameAsync(text)
        end)
        if ok and i then
            id = i
            name = text
        end
    end

    if not id then return nil, "User not found." end
    return { userId = id, name = name or tostring(id) }
end

local function findFavorite(userId)
    for i, f in ipairs(Favorites) do
        if f.userId == userId then return i, f end
    end
    return nil, nil
end

local function updateStar()
    if PreviewUserId and findFavorite(PreviewUserId) then
        StarBtn.Text = "★"
        StarBtn.BackgroundColor3 = Color3.fromRGB(240, 180, 40)
    else
        StarBtn.Text = "☆"
        StarBtn.BackgroundColor3 = Color3.fromRGB(35, 145, 220)
    end
end

local function setPreview(user)
    if not user then
        PreviewUserId = nil
        PreviewName = nil
        ThumbImg.Image = ""
        InfoLabel.Text = "No avatar loaded.\n\nType a username or UserId in the box below and press Enter."
        updateStar()
        return
    end

    PreviewUserId = user.userId
    PreviewName = user.name

    ThumbImg.Image = string.format(
        "rbxthumb://type=Avatar&id=%d&w=420&h=420",
        user.userId
    )

    InfoLabel.Text = string.format(
        "Loaded:\n@%s\n\nID: %d\n\nPress CHANGE AVATAR to apply.",
        user.name, user.userId
    )

    updateStar()
end

local function doLookupAndPreview(text)
    local user, err = resolveUser(text)
    if not user then
        StatusLabel.Text = "✗ " .. (err or "Not found.")
        setPreview(nil)
        return nil
    end
    setPreview(user)
    StatusLabel.Text = "✓ Found: @" .. user.name
    return user
end

    end
end)

Close.MouseButton1Click:Connect(function()
    removeAvatar()
    Gui:Destroy()
end)

LocalPlayer.CharacterAdded:Connect(function(newCharacter)
    removeAvatar()
    task.wait(0.15)
    if not newCharacter or not newCharacter.Parent then return end
    for _, instance in ipairs(newCharacter:GetDescendants()) do
        if instance:IsA("BasePart") then
            instance.LocalTransparencyModifier = 0
        elseif instance:IsA("Decal") or instance:IsA("Texture") then
            instance.Transparency = 0
        end
    end
end)

renderFavorites()
updateStar()
